

------------ Adding google provider is preety simple:------------------

refer this: https://next-auth.js.org/providers/google


1. Create credentials -> create OAuth client id-> webapp -> put urls:
```
http://localhost:3000

http://localhost:3000/api/auth/callback/google
```

2. copy GOOGLE Clientid and client secret, put them at .env file. and use them.

3.  the app/lib/auth.ts:
```
import CredentialsProvider from "next-auth/providers/credentials"
import { JWT } from "next-auth/jwt";
import { AdapterUser } from "next-auth/adapters";
import GoogleProvider from "next-auth/providers/google";


export const NEXTAUTH = {
    providers: [
        CredentialsProvider({
            name: 'Credentials',
            credentials: {
                username: { label: "Username", type: "text", placeholder: "jsmith" },
                password: { label: "Password", type: "password", placeholder:"Password" }
            },
            async authorize(credentials, req) {
                return {
                    id:"1",
                    name:"Vishnu"
                }
            }
        }),
        GoogleProvider({
            clientId: process.env.GOOGLE_CLIENT_ID || "",
            clientSecret: process.env.GOOGLE_CLIENT_SECRET || ""
          })
    ],
    secret:process.env.NEXTAUTH_SECRET,
    callbacks:{
        async session({
            session,token,user,
          }: {
            session: any;token: JWT;user: AdapterUser & { id: string }; 
          }){
            console.log(session)
            if (session.user) {
              session.user.id = token.sub ?? user.id; 
            }
            console.log(session)
            return session;
          }
    }
}

```
